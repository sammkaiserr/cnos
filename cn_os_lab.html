            char bit = data.charAt(i);

                consecutiveOnes++;
                if (consecutiveOnes == 5) {
                    frame.append('0');
                    consecutiveOnes = 0;
                }
            } else {
                consecutiveOnes = 0;
            }
        }
        frame.append(flag);
        return frame.toString();
    }

    public static void main(String[] args) {

        String frame = sendData(data);

        int i = 0;
        while (i < frame.length()) {

                        i++; // Skip the stuffed '0' bit
                        consecutiveOnes = 0;
                    }

                    message.append(bit);
                    consecutiveOnes = 0;
                }
                i++;
            }
        }
    }

import java.util.Scanner;

public class CRC {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        System.out.print("Enter data to be transmitted: ");
        String data = scanner.nextLine();

        System.out.print("Enter the Generating polynomial: ");
        String divisor = scanner.nextLine();
        int divisorLength = divisor.length();

        String paddedData = data + "0".repeat(divisorLength - 1);
        System.out.println("----------------------------------------");
        System.out.println("Data padded with n-1 zeros : " + paddedData);
        System.out.println("----------------------------------------");

        String remainder = divide(paddedData, divisor);
        System.out.println("CRC or Check value is : " + remainder);
        System.out.println("----------------------------------------");

        String encodedData = data + remainder;
        System.out.println("Final data to be sent : " + encodedData);
        System.out.println("----------------------------------------");

        System.out.print("Enter the received data: ");
        String receivedData = scanner.nextLine();
        System.out.println("-----------------------------");
        System.out.println("Data received: " + receivedData);

        String receivedRemainder = divide(receivedData, divisor);
        if (receivedRemainder.equals("0".repeat(divisorLength - 1))) {
            System.out.println("No error detected");
        } else {
            System.out.println("Error detected in received data");
        }
        scanner.close();
    }

    public static String divide(String data, String divisor) {
        int divisorLength = divisor.length();
        String tempData = data.substring(0, divisorLength);
        for (int i = divisorLength; i < data.length(); i++) {
            if (tempData.charAt(0) == '1') {
                tempData = xor(tempData, divisor) + data.charAt(i);
            } else {
                tempData = xor(tempData, "0".repeat(divisorLength)) + data.charAt(i);
            }
        }
        if (tempData.charAt(0) == '1') {
            tempData = xor(tempData, divisor);
        } else {
            tempData = xor(tempData, "0".repeat(divisorLength));
        }
        return tempData.substring(1);
    }

    public static String xor(String a, String b) {
        StringBuilder result = new StringBuilder();
        int length = b.length();
        for (int i = 0; i < length; i++) {
            result.append(a.charAt(i) == b.charAt(i) ? '0' : '1');
        }
        return result.toString();
    }
}

import java.util.Scanner;

public class HammingCodeConversion {

    static void generateHammingCode(int[] dataBits) {
        int[] hammingCode = new int[7]; // 4 data bits + 3 parity bits
        hammingCode[2] = dataBits[0];
        hammingCode[4] = dataBits[1];
        hammingCode[5] = dataBits[2];
        hammingCode[6] = dataBits[3];
        // Calculate parity bits
        hammingCode[0] = hammingCode[2] ^ hammingCode[4] ^ hammingCode[6]; // P1
        hammingCode[1] = hammingCode[2] ^ hammingCode[5] ^ hammingCode[6]; // P2
        hammingCode[3] = hammingCode[4] ^ hammingCode[5] ^ hammingCode[6]; // P4
        System.out.print("Generated Hamming Code: ");
        for (int i = 0; i < 7; i++) System.out.print(hammingCode[i] + " ");
        System.out.println();
    }

    static void detectAndCorrectError(int[] hammingCode) {
        int c1 = hammingCode[0] ^ hammingCode[2] ^ hammingCode[4] ^ hammingCode[6]; // P1
        int c2 = hammingCode[1] ^ hammingCode[2] ^ hammingCode[5] ^ hammingCode[6]; // P2
        int c4 = hammingCode[3] ^ hammingCode[4] ^ hammingCode[5] ^ hammingCode[6]; // P4
        int errorPosition = c1 + (c2 * 2) + (c4 * 4);
        if (errorPosition == 0) {
            System.out.println("No error detected.");
        } else {
            System.out.println("Error detected at position: " + errorPosition);
            hammingCode[errorPosition - 1] = hammingCode[errorPosition - 1] == 0 ? 1 : 0;
            System.out.print("Corrected Hamming Code: ");
            for (int i = 0; i < 7; i++) System.out.print(hammingCode[i] + " ");
            System.out.println();
        }
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int[] dataBits    = new int[4];
        int[] hammingCode = new int[7];
        System.out.println("Enter 4 data bits:");
        for (int i = 0; i < 4; i++) dataBits[i] = sc.nextInt();
        generateHammingCode(dataBits);
        System.out.println("Enter received Hamming Code (7 bits):");
        for (int i = 0; i < 7; i++) hammingCode[i] = sc.nextInt();
        detectAndCorrectError(hammingCode);
    }
}
CN-4 Dijkstra's Shortest Path Algorithm
▾
Graph.java
import java.util.*;

class Graph {
    class Node implements Comparable<Node> {
        int vertex, distance;
        public Node(int vertex, int distance) {
            this.vertex   = vertex;
            this.distance = distance;
        }
        @Override
        public int compareTo(Node other) { return this.distance - other.distance; }
    }

    private int vertices;
    private LinkedList<Node>[] adjacencyList;

    public Graph(int vertices) {
        this.vertices = vertices;
        adjacencyList = new LinkedList[vertices];
        for (int i = 0; i < vertices; i++) adjacencyList[i] = new LinkedList<>();
    }

    public void addEdge(int source, int destination, int weight) {
        adjacencyList[source].add(new Node(destination, weight));
        adjacencyList[destination].add(new Node(source, weight));
    }

    public void dijkstra(int source) {
        PriorityQueue<Node> pq = new PriorityQueue<>();
        int[] distances = new int[vertices];
        Arrays.fill(distances, Integer.MAX_VALUE);
        distances[source] = 0;
        pq.add(new Node(source, 0));
        boolean[] finalized = new boolean[vertices];

        while (!pq.isEmpty()) {
            Node curr = pq.poll();
            int u = curr.vertex;
            if (finalized[u]) continue;
            finalized[u] = true;
            for (Node neighbor : adjacencyList[u]) {
                int v = neighbor.vertex, w = neighbor.distance;
                if (!finalized[v] && distances[u] + w < distances[v]) {
                    distances[v] = distances[u] + w;
                    pq.add(new Node(v, distances[v]));
                }
            }
        }
        printShortestPaths(distances, source);
    }

    public void printShortestPaths(int[] distances, int source) {
        System.out.println("Shortest paths from vertex " + source + ":");
        for (int i = 0; i < distances.length; i++)
            System.out.println("Vertex " + i + " is at distance " + distances[i]);
    }

    public static void main(String[] args) {
        int vertices = 6;
        Graph graph = new Graph(vertices);
        graph.addEdge(0, 1, 4);
        graph.addEdge(0, 2, 2);
        graph.addEdge(1, 2, 1);
        graph.addEdge(1, 3, 5);
        graph.addEdge(2, 3, 8);
        graph.addEdge(2, 4, 10);
        graph.addEdge(3, 4, 2);
        graph.addEdge(3, 5, 6);
        graph.addEdge(4, 5, 3);
        graph.dijkstra(0);
    }
}

#include <stdio.h>

int main() {
    int n, i;
    int bt[20], wt[20], tat[20], p[20];
    int total_wt = 0, total_tat = 0;

    printf("Enter the number of processes: ");
    scanf("%d", &n);

    for (i = 0; i < n; i++) {
        printf("Enter burst time for process %d: ", i + 1);
        scanf("%d", &bt[i]);
        p[i] = i + 1;
    }

    wt[0] = 0;
    for (i = 1; i < n; i++) {
        wt[i] = wt[i - 1] + bt[i - 1];
        total_wt += wt[i];
    }

    for (i = 0; i < n; i++) {
        tat[i] = bt[i] + wt[i];
        total_tat += tat[i];
    }

    printf("\nProcess\tBurst Time\tWaiting Time\tTurnaround Time\n");
    for (i = 0; i < n; i++)
        printf("P%d\t%d\t\t%d\t\t%d\n", p[i], bt[i], wt[i], tat[i]);

    printf("\nAverage Waiting Time    = %.2f", (float)total_wt / n);
    printf("\nAverage Turnaround Time = %.2f\n", (float)total_tat / n);
    return 0;
}

#include <stdio.h>

int main() {
    int n, i, j;
    int burst[20], p[20], wait[20], turnaround[20];
    int total_wait = 0, total_turnaround = 0;

    printf("Enter number of processes: ");
    scanf("%d", &n);

    for (i = 0; i < n; i++) {
        printf("Enter burst time for process %d: ", i + 1);
        scanf("%d", &burst[i]);
        p[i] = i + 1;
    }

    // Bubble sort by burst time
    for (i = 0; i < n - 1; i++) {
        for (j = 0; j < n - i - 1; j++) {
            if (burst[j] > burst[j + 1]) {
                int temp = burst[j]; burst[j] = burst[j+1]; burst[j+1] = temp;
                    temp = p[j];     p[j]     = p[j+1];     p[j+1]     = temp;
            }
        }
    }

    wait[0] = 0;
    for (i = 1; i < n; i++) {
        wait[i] = 0;
        for (j = 0; j < i; j++) wait[i] += burst[j];
        total_wait += wait[i];
    }

    printf("\nProcess\tBurst Time\tWaiting Time\tTurnaround Time\n");
    for (i = 0; i < n; i++) {
        turnaround[i] = burst[i] + wait[i];
        total_turnaround += turnaround[i];
        printf("P%d\t%d\t\t%d\t\t%d\n", p[i], burst[i], wait[i], turnaround[i]);
    }

    printf("\nAverage Waiting Time    = %.2f", (float)total_wait / n);
    printf("\nAverage Turnaround Time = %.2f\n", (float)total_turnaround / n);
    return 0;
}

#include <stdio.h>

int main() {
    int n, remaining = 0, time = 0, smallest;
    int bt[20], at[20], rt[20];
    int i, end, wait[20], turnaround[20], total_wait = 0, total_turnaround = 0;

    printf("Enter number of processes: ");
    scanf("%d", &n);

    for (i = 0; i < n; i++) {
        printf("Enter arrival time and burst time for process %d: ", i + 1);
        scanf("%d%d", &at[i], &bt[i]);
        rt[i] = bt[i];
    }

    remaining = n;
    printf("\nProcess Execution Order:\n");

    while (remaining != 0) {
        smallest = -1;
        for (i = 0; i < n; i++) {
            if (at[i] <= time && rt[i] > 0) {
                if (smallest == -1 || rt[i] < rt[smallest]) smallest = i;
            }
        }
        if (smallest == -1) { time++; continue; }
        rt[smallest]--;
        time++;
        if (rt[smallest] == 0) {
            remaining--;
            end = time;
            wait[smallest]       = end - at[smallest] - bt[smallest];
            turnaround[smallest] = end - at[smallest];
            total_wait       += wait[smallest];
            total_turnaround += turnaround[smallest];
            printf("P%d ", smallest + 1);
        }
    }

    printf("\n\nProcess\tArrival\tBurst\tWaiting\tTurnaround\n");
    for (i = 0; i < n; i++)
        printf("P%d\t%d\t%d\t%d\t%d\n", i+1, at[i], bt[i], wait[i], turnaround[i]);

    printf("\nAverage Waiting Time    = %.2f", (float)total_wait / n);
    printf("\nAverage Turnaround Time = %.2f\n", (float)total_turnaround / n);
    return 0;
}

        smallest = -1;

int main() {
    int n, i, time = 0, tq, remain;
    int at[10], bt[10], rt[10], wt[10], tat[10];
    int flag = 0;
    float total_wt = 0, total_tat = 0;

    printf("Enter number of processes: ");
    scanf("%d", &n);
    remain = n;

    for (i = 0; i < n; i++) {
        printf("Enter arrival time and burst time for process P%d: ", i + 1);
        scanf("%d%d", &at[i], &bt[i]);
        rt[i] = bt[i];
    }
    printf("Enter time quantum: ");
    scanf("%d", &tq);

    printf("\nExecution Order:\n");
    while (remain != 0) {
        flag = 0;
        for (i = 0; i < n; i++) {
            if (at[i] <= time && rt[i] > 0) {
                flag = 1;
                if (rt[i] > tq) {
                    printf("P%d ", i + 1);
                    time += tq;
                    rt[i] -= tq;
                } else {
                    printf("P%d ", i + 1);
                    time += rt[i];
                    wt[i]  = time - at[i] - bt[i];
                    tat[i] = time - at[i];
                    rt[i]  = 0;
                    remain--;
                    total_wt  += wt[i];
                    total_tat += tat[i];
                }
            }
        }
        if (flag == 0) time++;

    printf("\n\nProcess\tArrival\tBurst\tWaiting\tTurnaround\n");

        printf("P%d\t%d\t%d\t%d\t%d\n", i+1, at[i], bt[i], wt[i], tat[i]);

                int canExecute = 1;
                for (j = 0; j < r; j++)
                    if (need[i][j] > available[j]) { canExecute = 0; break; }
                if (canExecute) {
                    for (j = 0; j < r; j++) available[j] += alloc[i][j];
                    safeSeq[count++] = i;
                    finish[i] = 1;
                    flag = 1;
                }
            }
        }
        if (!flag) { printf("\nSystem is NOT in a safe state.\n"); return 0; }
    }

    printf("\nSystem is in a safe state.\nSafe sequence: ");
    for (i = 0; i < p; i++) printf("P%d ", safeSeq[i]);
    printf("\nAvailable Resources After Execution:\n");
    for (i = 0; i < r; i++) printf("%d ", available[i]);
    printf("\n");
    return 0;
}

    printf("Enter the track numbers:\n");

    n++;

    // Bubble sort
    for (i = 0; i < n - 1; i++)
        for (j = 0; j < n - i - 1; j++)
            if (t[j] > t[j+1]) { temp = t[j]; t[j] = t[j+1]; t[j+1] = temp; }

    for (i = 0; i < n; i++) if (t[i] == h) { pos = i; break; }

    k = 0;
    for (i = pos; i >= 0;   i--) atr[k++] = t[i];
    for (i = pos+1; i < n;  i++) atr[k++] = t[i];

    for (i = 0; i < k - 1; i++) {
        d[i] = atr[i] > atr[i+1] ? atr[i]-atr[i+1] : atr[i+1]-atr[i];
        sum += d[i];
    }

    printf("\nOrder of head movement:\n");
    for (i = 0; i < k; i++) printf("%d ", atr[i]);
    printf("\n\nTotal head movement: %d", sum);
    printf("\nAverage head movement: %.2f\n", (float)sum / (k - 1));
    getch();
    return 0;
}

#include <stdlib.h>

int A = 0, Q = 0, M = 0, Q_1 = 0, count;

void display(int step) {
    printf("Step %d => A: %4d, Q: %4d, Q-1: %d\n", step, A, Q, Q_1);
}

int arithmetic_right_shift() {
    int sign = (A < 0) ? 1 : 0;
    Q_1 = Q & 1;
    Q = Q >> 1;
    if (A & 1) Q |= (1 << (sizeof(int)*8 - 1));
    A = A >> 1;
    if (sign) A |= (~((1 << (sizeof(int)*8 - 1)) - 1));
    return 0;
}

void booth(int multiplicand, int multiplier, int n) {
    A = 0; M = multiplicand; Q = multiplier; Q_1 = 0; count = n;
    printf("\nInitial Values => A: %4d, Q: %4d, M: %4d, Q-1: %d\n", A, Q, M, Q_1);
    display(0);
    for (int step = 1; count > 0; step++) {
        int Q0 = Q & 1;
        if      (Q0 == 1 && Q_1 == 0) { A = A - M; printf("A = A - M => A = %d\n", A); }
        else if (Q0 == 0 && Q_1 == 1) { A = A + M; printf("A = A + M => A = %d\n", A); }
        arithmetic_right_shift();
        count--;
        display(step);
    }
    printf("\nFinal Product (A:Q): %d\n", (A << n) | (Q & ((1 << n) - 1)));
}

}

void leftShift(int *A, int *Q, int n) {
    *A = ((*A << 1) | ((*Q >> (n - 1)) & 1)) & ((1 << n) - 1);
    *Q = (*Q << 1) & ((1 << n) - 1);
}

void restoringDivision(int dividend, int divisor, int n) {
    int A = 0, Q = dividend, M = divisor;
    printf("Initial Values:\nA = %d, Q = %d, M = %d\n", A, Q, M);

    for (int i = 0; i < n; i++) {
        leftShift(&A, &Q, n);
        printf("\nStep %d:\nAfter Left Shift => A = %d, Q = %d\n", i+1, A, Q);
        A = A - M;
        printf("After A - M => A = %d\n", A);
        if (A < 0) {
            Q = Q & (~1);
            A = A + M;
            printf("A is negative, restoring A => A = %d, Q = %d\n", A, Q);
        } else {
            Q = Q | 1;
            printf("A is positive, Q updated => Q = %d\n", Q);
        }
    }
    printf("\nFinal Quotient (Q): %d\n", Q);
    printf("Final Remainder (A): %d\n", A);
}
